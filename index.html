// sketch.ino String cmd = ""; const uint8_t PIN_OUT1 = 3; const uint8_t PIN_OUT2 = 4; const uint8_t PIN_OUT3 = 5; void setup() { Serial.begin(9600); pinMode(PIN_OUT1, OUTPUT); pinMode(PIN_OUT2, OUTPUT); pinMode(PIN_OUT3, OUTPUT); digitalWrite(PIN_OUT1, LOW); digitalWrite(PIN_OUT2, LOW); digitalWrite(PIN_OUT3, LOW); } void loop() { while (Serial.available()) { char c = Serial.read(); if (c == '\\n') { processCmd(cmd); cmd = ""; } else if (c != '\\r') { cmd += c; } } } void processCmd(String s) { s.trim(); if (s.length() == 0) return; Serial.print("Recu: "); Serial.println(s); // utile pour debug via moniteur série if (s == "START") { digitalWrite(PIN_OUT1, HIGH); } else if (s == "STOPALL") { digitalWrite(PIN_OUT1, LOW); digitalWrite(PIN_OUT2, LOW); digitalWrite(PIN_OUT3, LOW); } else if (s == "OUT2_ON") digitalWrite(PIN_OUT2, HIGH); else if (s == "OUT2_OFF") digitalWrite(PIN_OUT2, LOW); else if (s == "OUT3_ON") digitalWrite(PIN_OUT3, HIGH); else if (s == "OUT3_OFF") digitalWrite(PIN_OUT3, LOW); // else ignore }
